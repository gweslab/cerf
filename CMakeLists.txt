cmake_minimum_required(VERSION 3.25)
project(cerf LANGUAGES CXX)

add_subdirectory(libservicelocator)
set(CERFENGINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(libCerfEngine)
add_subdirectory(libCerfCore)

option(CERF_BUILD_QT "Build the cerf-qt frontend" OFF)
if(CERF_BUILD_QT)
    function(cerf_read_define out header name value_pattern)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header}")
        file(STRINGS "${header}" define_line REGEX "#define ${name} ")
        string(REGEX MATCH "#define ${name} +${value_pattern}" define_match "${define_line}")
        set(${out} "${CMAKE_MATCH_1}" PARENT_SCOPE)
    endfunction()
    function(cerf_read_string_define out header name)
        cerf_read_define(define_value "${header}" ${name} "\"([^\"]+)\"")
        set(${out} "${define_value}" PARENT_SCOPE)
    endfunction()
    function(cerf_read_int_define out header name)
        cerf_read_define(define_value "${header}" ${name} "([0-9]+)")
        set(${out} "${define_value}" PARENT_SCOPE)
    endfunction()
    set(CERF_EXECUTABLE_NAMES_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/cerf-qt/executable_names_text.h")
    set(CERF_INSTALL_ROOT_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/cerf-qt/install_root_files_text.h")
    set(CERF_INSTALL_LAYOUT_HEADER
        "${CMAKE_CURRENT_SOURCE_DIR}/libCerfCore/include/cerfcore/core/install_layout_text.h")
    set(CERF_RESOURCE_PATHS_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/cerf-qt/resource_paths_text.h")

    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(CERF_HOST_BRANCH_PY "${CMAKE_CURRENT_SOURCE_DIR}/tools/host_branch.py")
    set(CERF_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CERF_HOST_BRANCH_PY}")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CERF_HOST_BRANCH_PY}"
                            --cpp-header "${CERF_GENERATED_DIR}/host_branch_table.h"
                    RESULT_VARIABLE cerf_host_branch_rc)
    if(NOT cerf_host_branch_rc EQUAL 0)
        message(FATAL_ERROR "tools/host_branch.py --cpp-header failed (${cerf_host_branch_rc})")
    endif()
    if(MSVC)
        string(APPEND CMAKE_RC_FLAGS " /c65001")
    endif()
    if(WIN32)
        cerf_read_string_define(cerf_logo_asset "${CERF_RESOURCE_PATHS_HEADER}" CERF_LOGO_ASSET)
        cerf_read_string_define(cerf_error_asset "${CERF_RESOURCE_PATHS_HEADER}" CERF_ERROR_ASSET)
        set(cerf_logo_icon_file "${CMAKE_CURRENT_SOURCE_DIR}/assets/${cerf_logo_asset}.ico")
        set(cerf_error_icon_file "${CMAKE_CURRENT_SOURCE_DIR}/assets/${cerf_error_asset}.ico")
        file(CONFIGURE OUTPUT "${CERF_GENERATED_DIR}/asset_icon_files.h" @ONLY CONTENT
             "#define CERF_LOGO_ICON_FILE \"@cerf_logo_icon_file@\"\n#define CERF_ERROR_ICON_FILE \"@cerf_error_icon_file@\"\n")
    endif()
    add_subdirectory(cerf-qt)
    if(WIN32)
        add_subdirectory(launcher-stub-win32)
        add_subdirectory(uninstaller-win32)
    endif()
endif()
